isInScope : (Γ : Scope) → (t : Tm A) → Maybe (inScope Γ t)
isInScope Γ (var A x) = (A,x) ∈? Γ
isInScope Γ (app t u) = do
  p <- isInScope Γ t
  q <- isInScope Γ u
  Just (p, q)
isInScope Γ (lam{A}{B} f) = fresh λ x. case isInScope (Cons (A, #x) Γ) (f x) of
  Nothing -> <Nothing, x>
  Just (p : isInScope (Cons (A, #x) Γ) (f x)) ->
    rebind (λ p x. <Just p, x>) x p
